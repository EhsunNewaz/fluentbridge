# FluentBridge - Database Schema Only

## 🗄️ **Core Database Tables**

### **Authentication & User Management**
```sql
-- Users table (extends Supabase auth.users)
CREATE TABLE public.users (
  id UUID REFERENCES auth.users(id) PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  full_name VARCHAR(255) NOT NULL,
  bengali_name VARCHAR(255),
  phone VARCHAR(20),
  location VARCHAR(100) DEFAULT 'Khulna, Bangladesh',
  preferred_language VARCHAR(10) DEFAULT 'bn',
  date_of_birth DATE,
  education_level VARCHAR(50),
  target_band_score DECIMAL(2,1),
  current_level VARCHAR(20),
  profile_image_url TEXT,
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- User roles
CREATE TABLE user_roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  role VARCHAR(50) NOT NULL, -- 'student', 'instructor', 'admin'
  granted_at TIMESTAMPTZ DEFAULT NOW(),
  is_active BOOLEAN DEFAULT true,
  UNIQUE(user_id, role)
);
```

### **Learning Content Hierarchy (4 Levels)**
```sql
-- Learning Tracks (IELTS Prep, Grammar Foundations, Phonics Foundation)
CREATE TABLE learning_tracks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(200) NOT NULL,
  slug VARCHAR(200) UNIQUE NOT NULL,
  description TEXT,
  bengali_description TEXT,
  level VARCHAR(20) CHECK (level IN ('foundation', 'beginner', 'intermediate', 'advanced')),
  order_index INTEGER DEFAULT 0,
  is_active BOOLEAN DEFAULT true,
  estimated_duration_hours INTEGER,
  is_free BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Courses within tracks
CREATE TABLE courses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  track_id UUID REFERENCES learning_tracks(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  slug VARCHAR(200) NOT NULL,
  description TEXT,
  bengali_description TEXT,
  level VARCHAR(20),
  estimated_hours INTEGER,
  price DECIMAL(10,2) DEFAULT 0,
  is_free BOOLEAN DEFAULT true,
  order_index INTEGER DEFAULT 0,
  instructor_id UUID REFERENCES users(id),
  is_published BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(track_id, slug)
);

-- Modules within courses
CREATE TABLE course_modules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  slug VARCHAR(200) NOT NULL,
  description TEXT,
  bengali_description TEXT,
  order_index INTEGER DEFAULT 0,
  estimated_duration_minutes INTEGER,
  learning_objectives TEXT[],
  bengali_objectives TEXT[],
  is_published BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(course_id, slug)
);

-- Lessons within modules
CREATE TABLE lessons (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  module_id UUID REFERENCES course_modules(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  slug VARCHAR(200) NOT NULL,
  description TEXT,
  content JSONB,
  bengali_content JSONB,
  lesson_type VARCHAR(50) CHECK (lesson_type IN ('video', 'reading', 'interactive', 'quiz')),
  order_index INTEGER DEFAULT 0,
  estimated_duration_minutes INTEGER,
  difficulty_level INTEGER CHECK (difficulty_level BETWEEN 0 AND 5),
  is_published BOOLEAN DEFAULT false,
  is_free BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(module_id, slug)
);
```

### **Phonics Foundation System**
```sql
-- Phonics sounds database
CREATE TABLE phonics_sounds (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  english_sound VARCHAR(10) NOT NULL,
  ipa_symbol VARCHAR(10) NOT NULL,
  bengali_equivalent VARCHAR(20),
  bengali_explanation TEXT,
  common_mistakes TEXT[],
  practice_words TEXT[],
  audio_example_url TEXT,
  difficulty_level INTEGER CHECK (difficulty_level BETWEEN 1 AND 5),
  sound_category VARCHAR(50), -- 'vowel', 'consonant', 'diphthong'
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- User phonics progress
CREATE TABLE user_phonics_progress (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  sound_id UUID REFERENCES phonics_sounds(id) ON DELETE CASCADE,
  mastery_level INTEGER DEFAULT 0 CHECK (mastery_level BETWEEN 0 AND 5),
  practice_sessions INTEGER DEFAULT 0,
  last_practiced_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, sound_id)
);
```

### **Assessment System (6-Part)**
```sql
-- Assessment types
CREATE TABLE assessments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(200) NOT NULL,
  slug VARCHAR(200) UNIQUE NOT NULL,
  type VARCHAR(50) CHECK (type IN ('phonics_diagnostic', 'overall_ielts', 'reading', 'writing', 'listening', 'speaking')),
  description TEXT,
  bengali_description TEXT,
  duration_minutes INTEGER NOT NULL,
  total_points INTEGER DEFAULT 100,
  skill_categories TEXT[],
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Assessment questions
CREATE TABLE assessment_questions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
  question_text TEXT NOT NULL,
  bengali_hint TEXT,
  question_type VARCHAR(50) CHECK (question_type IN ('multiple_choice', 'true_false', 'fill_blank', 'drag_drop')),
  options JSONB,
  correct_answer JSONB,
  explanation TEXT,
  bengali_explanation TEXT,
  points INTEGER DEFAULT 1,
  skill_category VARCHAR(100),
  difficulty_level INTEGER CHECK (difficulty_level BETWEEN 0 AND 5),
  order_index INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- User assessment results
CREATE TABLE user_assessment_results (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
  started_at TIMESTAMPTZ DEFAULT NOW(),
  completed_at TIMESTAMPTZ,
  total_score INTEGER,
  percentage_score DECIMAL(5,2),
  time_taken_minutes INTEGER,
  skill_breakdown JSONB,
  recommendations JSONB,
  band_prediction DECIMAL(2,1),
  is_completed BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

### **Progress Tracking**
```sql
-- Course enrollments
CREATE TABLE course_enrollments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
  enrolled_at TIMESTAMPTZ DEFAULT NOW(),
  completed_at TIMESTAMPTZ,
  progress_percentage DECIMAL(5,2) DEFAULT 0,
  current_lesson_id UUID REFERENCES lessons(id),
  time_spent_minutes INTEGER DEFAULT 0,
  is_active BOOLEAN DEFAULT true,
  UNIQUE(user_id, course_id)
);

-- Lesson progress
CREATE TABLE user_lesson_progress (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  lesson_id UUID REFERENCES lessons(id) ON DELETE CASCADE,
  status VARCHAR(20) CHECK (status IN ('not_started', 'in_progress', 'completed')) DEFAULT 'not_started',
  completion_date TIMESTAMPTZ,
  score INTEGER,
  time_spent_seconds INTEGER DEFAULT 0,
  difficulty_rating INTEGER CHECK (difficulty_rating BETWEEN 1 AND 5),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, lesson_id)
);
```

### **Community Features**
```sql
-- Study groups
CREATE TABLE study_groups (
```

### **Feedback & Communication**
```sql
-- Lesson feedback system
CREATE TABLE lesson_feedback (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  lesson_id UUID REFERENCES lessons(id) ON DELETE CASCADE,
  feedback_type VARCHAR(50) CHECK (feedback_type IN ('question', 'improvement_suggestion', 'difficulty_feedback')),
  content TEXT NOT NULL,
  bengali_content TEXT,
  difficulty_rating INTEGER CHECK (difficulty_rating BETWEEN 1 AND 5),
  is_public BOOLEAN DEFAULT false,
  status VARCHAR(20) DEFAULT 'open', -- 'open', 'answered', 'implemented'
  admin_response TEXT,
  bengali_response TEXT,
  responded_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Mock test system
CREATE TABLE mock_tests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(200) NOT NULL,
  description TEXT,
  test_type VARCHAR(50), -- 'full_ielts', 'academic_reading', 'general_writing'
  duration_minutes INTEGER,
  sections JSONB, -- Test structure
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Mock test attempts
CREATE TABLE mock_test_attempts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  mock_test_id UUID REFERENCES mock_tests(id) ON DELETE CASCADE,
  started_at TIMESTAMPTZ DEFAULT NOW(),
  completed_at TIMESTAMPTZ,
  answers JSONB,
  results JSONB,
  band_scores JSONB, -- Overall and section band scores
  time_taken_minutes INTEGER,
  is_completed BOOLEAN DEFAULT false
);
```

### **Essential Indexes**
```sql
-- User lookup optimization
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_active ON users(is_active) WHERE is_active = true;

-- Content hierarchy navigation
CREATE INDEX idx_courses_track ON courses(track_id, order_index);
CREATE INDEX idx_modules_course ON course_modules(course_id, order_index);
CREATE INDEX idx_lessons_module ON lessons(module_id, order_index);

-- Progress tracking optimization
CREATE INDEX idx_lesson_progress_user ON user_lesson_progress(user_id, status);
CREATE INDEX idx_enrollments_user ON course_enrollments(user_id, is_active);

-- Assessment performance
CREATE INDEX idx_assessment_results_user ON user_assessment_results(user_id, completed_at DESC);
CREATE INDEX idx_phonics_progress_user ON user_phonics_progress(user_id, mastery_level);

-- Community features
CREATE INDEX idx_study_groups_active ON study_groups(is_active, target_band_min, target_band_max);
CREATE INDEX idx_feedback_status ON lesson_feedback(status, created_at DESC);
```

### **Row Level Security (RLS)**
```sql
-- Enable RLS on sensitive tables
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE course_enrollments ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_lesson_progress ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_assessment_results ENABLE ROW LEVEL SECURITY;

-- Students can only see their own data
CREATE POLICY student_own_data ON user_lesson_progress
  FOR ALL USING (auth.uid() = user_id);

CREATE POLICY student_own_results ON user_assessment_results
  FOR ALL USING (auth.uid() = user_id);
```
